//Pascals Triangle i
//First Approach
//Time Complexity-O(c-1)
//Space Complexity-O(1)
class Solution {
    public int pascalTriangleI(int r, int c) {
        long res=1;
        for(int i=0;i<c-1;i++)
        {
            res=res*(r-1-i);
            res=res/(i+1);
        }
        
        return (int)res;
            }
}
